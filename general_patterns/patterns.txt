        cin >> a[i].first;
        cin >> a[i].second;
 
    }
    map<ll,int>v;
 
    for(ll i = 0 ; i < n ;  i++) {
        v[a[i].first] +=1;
        v[a[i].second] -=1;// we are adding and
        //putting it equal coz same year 2 people
        //can born/die so events mights be repeated
        
    }
    ll cnt =0 ;
    ll best =INT_MIN;
    ll year;
    for(auto  &i: v) {
        cnt += i.second;
        if(cnt > best) {
            best= cnt ;
            year = i.first;
        }
    }
    cout << year<< " "<< best << endl;
